import fs from 'node:fs/promises';import path from 'node:path';import {randomUUID}from'node:crypto';
import {StorageClient}from'./src/services/workers/storage-client.js';import{InstallerService}from'./src/services/install/installer.js';import{OfficialMetadataReader}from'./src/services/catalog/transport.js';import{CatalogService}from'./src/services/catalog/service.js';import{InstallMetadataService}from'./src/services/catalog/install-metadata.js';
const root=await fs.mkdtemp(path.resolve('.test-build/final-vanilla-'));await fs.mkdir(path.join(root,'metadata'));await fs.mkdir(path.join(root,'games'));const entry=path.resolve('dist/workers/entry.cjs');const storage=await StorageClient.open(entry,path.join(root,'metadata'));const rootId=randomUUID();await storage.createInstallation({rootId,dataDirectory:path.join(root,'games')});const reader=new OfficialMetadataReader();const service=new InstallerService(storage,entry,new InstallMetadataService(new CatalogService(reader,storage),reader));
try{await service.initialize();const task=await service.create(randomUUID(),{rootId,mcVersionId:'1.20.1',displayName:'Final vanilla verification',isolation:'on'});let prior='';for(;;){const current=await service.task(rootId,task.taskId);if(prior!==current.stage){console.log(JSON.stringify({workspace:root,stage:current.stage,state:current.task.execution.state}));prior=current.stage;}if(['completed','failed','cancelled','needsRecovery'].includes(current.task.execution.state)){if(current.task.execution.state!=='completed')throw Error('OFFICIAL_INSTALL_FAILED');break;}await new Promise(r=>setTimeout(r,1000));}}finally{await service.close();await storage.close();}
